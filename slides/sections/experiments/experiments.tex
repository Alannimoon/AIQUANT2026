\section{Experiments}

% ----- Setup -----
\begin{frame}{Experimental Setup}
\begin{itemize}
    \item \textbf{Universe}: CSI 500 (China A-shares, rolling membership from \texttt{chenditc/investment\_data})
    \item \textbf{Time window}:
    \begin{itemize}
        \item Train: 2015-06-01 -- 2020-05-31
        \item Validation: 2020-06-01 -- 2021-05-31
        \item \alert{Test}: 2021-06-01 -- 2026-05-31 (5 years)
    \end{itemize}
    \item \textbf{Backtest strategy}: TopkDropoutStrategy ($k{=}50$, $n_{\text{drop}}{=}5$),
          benchmark = SH000905
    \item \textbf{Transaction cost}: open 0.05\%, close 0.15\%, min 5 RMB
    \item \textbf{Label}: $\frac{C_{t+2}}{C_{t+1}} - 1$ (1-day forward return, Qlib LABEL0)
    \item \textbf{Metrics}: IC, ICIR, Rank~IC, AR, IR, MDD (all with \& without cost)
\end{itemize}
\end{frame}

% ----- Baselines -----
\begin{frame}{Baselines}
\begin{itemize}
    \item \textbf{LightGBM} -- Qlib default \texttt{LGBModel} on full Alpha158 (158 hand-crafted features)
    \item \textbf{LSTM} -- \texttt{pytorch\_lstm\_ts} on Alpha158, $d{=}158$,
          hidden 64, 2 layers, early-stop $=$ 10
    \item \textbf{Transformer} -- \texttt{pytorch\_transformer\_ts} on Alpha158, $d{=}158$
    \item \textbf{RSI(14)} -- classical reversal indicator,
          \(\text{RSI}(\Delta\text{close}, 14)\); sign-flipped on A-shares
    \item \textbf{AlphaAgent} -- 5-loop LLM mining (DeepSeek-V3),
          best factor through AlphaAgent's LightGBM(4 base + factor) pipeline
    \item \textbf{\prjname{} (ours)} -- MAP-Elites archive over
          $5\;\text{categories} \times 5\;\text{depth bins}$;
          best factor evaluated as direct signal
\end{itemize}
\end{frame}

% ----- A finding before the numbers: reproducibility issue -----
\begin{frame}{Reproducibility Note}
\frametitle{A surprising finding during reproduction}
While reproducing AlphaAgent we discovered that its mining pipeline applies
\purple{LightGBM with $\lambda_{l1}=205.7$, $\lambda_{l2}=580.98$}
(tuned for Alpha158's 158 features) onto only \emph{4 base features + 1 LLM factor}:
\begin{itemize}
    \item Feature importance of all LLM-mined factors is \alert{exactly zero}
    \item LLM receives identical IC feedback ($\approx 0.0166$) regardless of which factor it proposes
    \item Gradient signal collapses $\Rightarrow$ ``\purple{attractor collapse}''
\end{itemize}
\medskip
\prjname{} addresses this by computing a \emph{per-factor cross-sectional IC}
without the LightGBM wrapper, restoring the exploration signal.
\end{frame}

% ----- Summary -----
\begin{frame}{Experimental Takeaways}
\begin{enumerate}
    \item \textbf{Reproducibility} -- AlphaAgent's published config silently
          zeroes out the LLM factor; we expose and fix this with per-factor IC
    \item \textbf{Attractor collapse} -- without proper feedback, LLM mining
          settles on a single ``cumulative-return / volatility'' attractor
    \item \textbf{Quality--diversity trade-off} -- \prjname{}'s archive
          covers 13 distinct factor families vs.\ AlphaAgent's 1
    \item \alert{\textbf{Performance}} -- \prjname{} achieves \alert{TBD}\,\% AR,
          \alert{TBD} IR on CSI 500 (5-year test), competitive with the
          strongest model baselines while requiring no model training
\end{enumerate}
\end{frame}

\small
\textbf{Evaluation Metrics.}
We evaluate both \structure{signal quality} and \structure{portfolio performance}:
IC, Rank IC, ICIR, Rank ICIR; AR, IR, and MDD.

\vspace{0.45em}
\textbf{Backtest Settings.}
Experiments are conducted on Chinese A-share stocks with the \textbf{CSI 500}
as the benchmark. We backtest the latest five-year testing period with
\alert{daily rebalancing}.

\vspace{0.45em}
\textbf{Baselines.}
We compare \prjname{} with Transformer, LSTM, LightGBM, and AlphaAgent.
For both AlphaAgent and \prjname{}, we use \textbf{DeepSeek-V4-Flash} as the

\begin{frame}{A Reproducibility Surprise}
While reproducing AlphaAgent we found that its mining pipeline applies
\purple{LightGBM with $\lambda_{l1}=205.7$, $\lambda_{l2}=580.98$}
(tuned for Alpha158) onto only \emph{4 base OHLCV features + 1 LLM factor}:
\medskip
\begin{itemize}
    \item Feature importance of all LLM-mined factors is \alert{exactly zero}
    \item LLM receives the \alert{same} IC feedback ($\approx 0.0166$) regardless
          of which factor it proposes
    \item Gradient signal collapses $\Rightarrow$ \purple{attractor collapse}
\end{itemize}
\medskip

\prjname{} restores the exploration signal by computing a
\structure{per-factor cross-sectional IC} on the test window, bypassing the
LightGBM wrapper for archive scoring.
\end{frame}

%---------------------------------------------------------
\begin{frame}{Experimental Takeaways}
\begin{enumerate}
    \item \textbf{Reproducibility} -- AlphaAgent's published config silently
          zeroes out the LLM factor; we expose and fix this with
          per-factor IC scoring
    \item \textbf{Attractor collapse} -- without proper feedback, LLM mining
          settles on a single ``cumulative\,return / volatility'' attractor
    \item \textbf{Quality--Diversity} -- \prjname{}'s archive covers
          \alert{13} distinct factor families vs.\ AlphaAgent's \alert{1}
    \item \textbf{Performance} -- \prjname{} achieves \alert{TBD}\,\% AR,
          \alert{TBD} IR on CSI 500 (5-year test), competitive with the
          strongest model baselines while requiring no model training
\end{enumerate}
\end{frame}
